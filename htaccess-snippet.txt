# Tắt liệt kê thư mục
<IfModule mod_autoindex.c>
    Options -Indexes
</IfModule>

# Chặn tải trực tiếp các file cấu hình / dữ liệu / tài liệu nội bộ
<FilesMatch "^(config\.register(\.local)?(\.sample)?\.php|\.gitignore|htaccess-snippet\.txt|README\.md|.*\.sql)$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</FilesMatch>
