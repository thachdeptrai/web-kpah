# ==========================================================================
# KPAH hardening – GỘP vào .htaccess gốc ĐANG CÓ, không ghi đè toàn bộ file.
# Kiểm tra các route hiện tại sau khi thêm. Yêu cầu Apache + mod_rewrite.
# Thư mục includes/, storage/, database/ đã có .htaccess riêng.
# ==========================================================================

# Tắt liệt kê thư mục
<IfModule mod_autoindex.c>
    Options -Indexes
</IfModule>

<IfModule mod_rewrite.c>
    RewriteEngine On
    # Chặn dotfiles và các thư mục nội bộ/repository
    RewriteRule (^|/)\\. - [F,L]
    RewriteRule ^(includes|storage|database|\\.git)(/|$) - [F,L,NC]
</IfModule>

# Chặn tải cấu hình, thông tin debug, backup, log, SQL và tài liệu nội bộ
<FilesMatch "(?i)^(\\.env.*|\\.gitignore|\\.htaccess|\\.htpasswd|.*\\.(sql|log|ini|ya?ml|dist|bak|backup|old|orig|swp|tmp|dump|zip|tar|gz|7z|md))$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</FilesMatch>

<FilesMatch "(?i)^(config\\.php|config\\.register(\\.local|\\.local\\.sample|\\.sample)?\\.php|check\\.php|captcha\\.php)$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order allow,deny
        Deny from all
    </IfModule>
</FilesMatch>
